# 前端行为追踪集成示例

## 1. 小程序端集成

### 1.1 基础行为追踪器

```javascript
// utils/behavior-tracker.js

/**
 * 行为追踪器
 * 自动收集用户在小程序中的行为数据
 */
class BehaviorTracker {
  constructor() {
    this.sessionId = this.generateSessionId();
    this.userId = this.getUserId();
    this.startTime = Date.now();
    this.pageStartTime = Date.now();
    this.currentPage = '';
    
    // 初始化追踪
    this.initTracking();
  }

  /**
   * 生成会话ID
   */
  generateSessionId() {
    return `session_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  }

  /**
   * 获取用户ID
   */
  getUserId() {
    // 从本地存储或全局状态获取用户ID
    const userInfo = wx.getStorageSync('userInfo');
    return userInfo ? userInfo.user_id : 'anonymous';
  }

  /**
   * 获取设备信息
   */
  getDeviceInfo() {
    const systemInfo = wx.getSystemInfoSync();
    return {
      platform: systemInfo.platform,
      version: systemInfo.version,
      screenWidth: systemInfo.screenWidth,
      screenHeight: systemInfo.screenHeight,
      pixelRatio: systemInfo.pixelRatio,
      language: systemInfo.language,
      model: systemInfo.model
    };
  }

  /**
   * 初始化追踪
   */
  initTracking() {
    this.trackPageView();
    this.trackUserInteractions();
    this.trackAppLifecycle();
  }

  /**
   * 页面访问追踪
   */
  trackPageView() {
    const pages = getCurrentPages();
    if (pages.length > 0) {
      const currentPage = pages[pages.length - 1];
      const pagePath = currentPage.route;
      
      if (pagePath !== this.currentPage) {
        // 记录上一个页面的停留时长
        if (this.currentPage) {
          this.trackEvent('page_leave', '页面离开', {
            page_path: this.currentPage,
            stay_duration: Date.now() - this.pageStartTime
          });
        }

        // 记录新页面访问
        this.currentPage = pagePath;
        this.pageStartTime = Date.now();
        
        this.trackEvent('page_view', '页面访问', {
          page_path: pagePath,
          referrer: this.getReferrer()
        });
      }
    }
  }

  /**
   * 获取来源页面
   */
  getReferrer() {
    const pages = getCurrentPages();
    if (pages.length > 1) {
      return pages[pages.length - 2].route;
    }
    return '';
  }

  /**
   * 用户交互追踪
   */
  trackUserInteractions() {
    // 监听页面点击事件
    wx.onTouchStart((e) => {
      this.trackEvent('click', '点击事件', {
        element_id: e.target.id || e.target.className,
        click_position: { x: e.touches[0].clientX, y: e.touches[0].clientY }
      });
    });

    // 监听页面滚动事件
    let scrollTimer;
    wx.onPageScroll((e) => {
      clearTimeout(scrollTimer);
      scrollTimer = setTimeout(() => {
        this.trackEvent('scroll', '滚动事件', {
          scroll_top: e.scrollTop,
          scroll_depth: this.calculateScrollDepth(e.scrollTop)
        });
      }, 1000);
    });
  }

  /**
   * 计算滚动深度
   */
  calculateScrollDepth(scrollTop) {
    const systemInfo = wx.getSystemInfoSync();
    const pageHeight = systemInfo.windowHeight;
    return Math.min(scrollTop / pageHeight, 1);
  }

  /**
   * 应用生命周期追踪
   */
  trackAppLifecycle() {
    // 应用启动
    wx.onAppShow(() => {
      this.trackEvent('app_show', '应用启动');
    });

    // 应用隐藏
    wx.onAppHide(() => {
      this.trackEvent('app_hide', '应用隐藏', {
        session_duration: Date.now() - this.startTime
      });
    });
  }

  /**
   * 商品相关事件追踪
   */
  trackProductEvents(productId, productName) {
    // 商品浏览
    this.trackEvent('product_view', '商品浏览', {
      product_id: productId,
      product_name: productName
    });
  }

  /**
   * 购物车事件追踪
   */
  trackCartEvents(productId, action, quantity = 1) {
    this.trackEvent(action, action === 'add_to_cart' ? '加入购物车' : '从购物车移除', {
      product_id: productId,
      quantity: quantity
    });
  }

  /**
   * 订单事件追踪
   */
  trackOrderEvents(orderId, orderAmount, action) {
    this.trackEvent(action, this.getOrderEventName(action), {
      order_id: orderId,
      order_amount: orderAmount
    });
  }

  /**
   * 获取订单事件名称
   */
  getOrderEventName(action) {
    const eventNames = {
      'order_create': '创建订单',
      'order_pay': '订单支付',
      'order_cancel': '订单取消',
      'order_complete': '订单完成'
    };
    return eventNames[action] || action;
  }

  /**
   * 搜索事件追踪
   */
  trackSearchEvents(keyword, resultCount) {
    this.trackEvent('search', '搜索', {
      keyword: keyword,
      result_count: resultCount
    });
  }

  /**
   * 上报事件数据
   */
  trackEvent(eventType, eventName, properties = {}) {
    const eventData = {
      event_type: eventType,
      event_name: eventName,
      session_id: this.sessionId,
      user_id: this.userId,
      page_path: this.currentPage,
      properties: properties,
      device_info: this.getDeviceInfo(),
      timestamp: Date.now()
    };

    // 发送到后端
    this.sendToServer(eventData);
  }

  /**
   * 发送数据到服务器
   */
  sendToServer(eventData) {
    wx.request({
      url: 'https://your-api-domain.com/api/v1/behavior/track',
      method: 'POST',
      data: eventData,
      header: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${wx.getStorageSync('token')}`
      },
      success: (res) => {
        if (res.statusCode === 200) {
          console.log('行为数据上报成功:', res.data);
        } else {
          console.error('行为数据上报失败:', res);
        }
      },
      fail: (err) => {
        console.error('行为数据上报失败:', err);
        // 失败时存储到本地，稍后重试
        this.storeFailedEvent(eventData);
      }
    });
  }

  /**
   * 存储失败的事件数据
   */
  storeFailedEvent(eventData) {
    const failedEvents = wx.getStorageSync('failedBehaviorEvents') || [];
    failedEvents.push({
      ...eventData,
      retry_count: 0,
      created_at: Date.now()
    });
    wx.setStorageSync('failedBehaviorEvents', failedEvents);
  }

  /**
   * 重试失败的事件
   */
  retryFailedEvents() {
    const failedEvents = wx.getStorageSync('failedBehaviorEvents') || [];
    if (failedEvents.length === 0) return;

    const maxRetries = 3;
    const retryEvents = [];

    failedEvents.forEach(event => {
      if (event.retry_count < maxRetries) {
        event.retry_count++;
        retryEvents.push(event);
        this.sendToServer(event);
      }
    });

    wx.setStorageSync('failedBehaviorEvents', retryEvents);
  }
}

// 创建全局行为追踪器实例
const behaviorTracker = new BehaviorTracker();

// 导出供其他模块使用
module.exports = behaviorTracker;
```

### 1.2 页面集成示例

```javascript
// pages/product/detail.js

const behaviorTracker = require('../../utils/behavior-tracker.js');

Page({
  data: {
    product: null,
    loading: true
  },

  onLoad(options) {
    const productId = options.id;
    this.loadProductDetail(productId);
    
    // 追踪商品详情页访问
    behaviorTracker.trackProductEvents(productId, '商品详情页');
  },

  onShow() {
    // 页面显示时更新页面追踪
    behaviorTracker.trackPageView();
  },

  onHide() {
    // 页面隐藏时记录停留时长
    behaviorTracker.trackEvent('page_leave', '页面离开', {
      page_path: 'pages/product/detail',
      stay_duration: Date.now() - behaviorTracker.pageStartTime
    });
  },

  // 加载商品详情
  loadProductDetail(productId) {
    wx.request({
      url: `https://your-api-domain.com/api/v1/product/${productId}`,
      success: (res) => {
        this.setData({
          product: res.data.data,
          loading: false
        });
      }
    });
  },

  // 加入购物车
  addToCart() {
    const product = this.data.product;
    
    wx.request({
      url: 'https://your-api-domain.com/api/v1/cart/add',
      method: 'POST',
      data: {
        product_id: product.id,
        quantity: 1
      },
      success: (res) => {
        if (res.statusCode === 200) {
          wx.showToast({
            title: '已加入购物车',
            icon: 'success'
          });
          
          // 追踪加入购物车事件
          behaviorTracker.trackCartEvents(product.id, 'add_to_cart', 1);
        }
      }
    });
  },

  // 立即购买
  buyNow() {
    const product = this.data.product;
    
    // 追踪立即购买事件
    behaviorTracker.trackEvent('buy_now', '立即购买', {
      product_id: product.id,
      product_name: product.name
    });

    // 跳转到订单确认页
    wx.navigateTo({
      url: `/pages/order/confirm?product_id=${product.id}&quantity=1`
    });
  },

  // 收藏商品
  toggleFavorite() {
    const product = this.data.product;
    const isFavorite = this.data.isFavorite;
    
    const action = isFavorite ? 'remove_from_favorite' : 'add_to_favorite';
    const eventName = isFavorite ? '取消收藏' : '添加收藏';
    
    // 追踪收藏事件
    behaviorTracker.trackEvent(action, eventName, {
      product_id: product.id,
      product_name: product.name
    });

    // 更新收藏状态
    this.setData({
      isFavorite: !isFavorite
    });
  },

  // 分享商品
  onShareAppMessage() {
    const product = this.data.product;
    
    // 追踪分享事件
    behaviorTracker.trackEvent('product_share', '分享商品', {
      product_id: product.id,
      product_name: product.name
    });

    return {
      title: product.name,
      path: `/pages/product/detail?id=${product.id}`,
      imageUrl: product.images[0]
    };
  }
});
```

### 1.3 购物车页面集成

```javascript
// pages/cart/index.js

const behaviorTracker = require('../../utils/behavior-tracker.js');

Page({
  data: {
    cartItems: [],
    totalAmount: 0
  },

  onLoad() {
    this.loadCartItems();
  },

  onShow() {
    behaviorTracker.trackPageView();
  },

  // 加载购物车商品
  loadCartItems() {
    wx.request({
      url: 'https://your-api-domain.com/api/v1/cart/list',
      success: (res) => {
        this.setData({
          cartItems: res.data.data,
          totalAmount: res.data.total_amount
        });
      }
    });
  },

  // 更新商品数量
  updateQuantity(e) {
    const { index, type } = e.currentTarget.dataset;
    const item = this.data.cartItems[index];
    const newQuantity = type === 'increase' ? item.quantity + 1 : item.quantity - 1;

    if (newQuantity <= 0) {
      // 移除商品
      this.removeFromCart(item.id);
    } else {
      // 更新数量
      this.updateCartItem(item.id, newQuantity);
    }
  },

  // 移除商品
  removeFromCart(itemId) {
    wx.request({
      url: `https://your-api-domain.com/api/v1/cart/remove/${itemId}`,
      method: 'DELETE',
      success: (res) => {
        if (res.statusCode === 200) {
          this.loadCartItems();
          
          // 追踪移除购物车事件
          behaviorTracker.trackCartEvents(itemId, 'remove_from_cart');
        }
      }
    });
  },

  // 结算
  checkout() {
    // 追踪结算事件
    behaviorTracker.trackEvent('cart_checkout', '购物车结算', {
      total_amount: this.data.totalAmount,
      item_count: this.data.cartItems.length
    });

    wx.navigateTo({
      url: '/pages/order/confirm'
    });
  }
});
```

// pages/order/confirm.js

const behaviorTracker = require('../../utils/behavior-tracker.js');

Page({
  data: {
    orderInfo: null,
    address: null
  },

  onLoad(options) {
    this.loadOrderInfo(options);
  },

  onShow() {
    behaviorTracker.trackPageView();
  },

  // 加载订单信息
  loadOrderInfo(options) {
    wx.request({
      url: 'https://your-api-domain.com/api/v1/order/preview',
      data: options,
      success: (res) => {
        this.setData({
          orderInfo: res.data.data
        });
      }
    });
  },

  // 提交订单
  submitOrder() {
    const orderInfo = this.data.orderInfo;
    
    wx.request({
      url: 'https://your-api-domain.com/api/v1/order/create',
      method: 'POST',
      data: orderInfo,
      success: (res) => {
        if (res.statusCode === 200) {
          const orderId = res.data.data.order_id;
          
          // 追踪创建订单事件
          behaviorTracker.trackOrderEvents(
            orderId, 
            orderInfo.total_amount, 
            'order_create'
          );

          // 跳转到支付页面
          wx.navigateTo({
            url: `/pages/payment/index?order_id=${orderId}`
          });
        }
      }
    });
  }
});
